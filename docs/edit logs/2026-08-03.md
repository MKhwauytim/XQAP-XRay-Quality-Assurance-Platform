function groupRows(
  rows: ExecutiveReportRow[],
  keyFor: (row: ExecutiveReportRow) => string,
): Map<string, ExecutiveReportRow[]> {
  const groups = new Map<string, ExecutiveReportRow[]>();
  for (const row of rows) {
    const key = keyFor(row);
    const bucket = groups.get(key);
    if (bucket) {
      bucket.push(row);
    } else {
      groups.set(key, [row]);
    }
  }
  return groups;
}

## v59.125 — 2026-08-03 — Add (storage): shared bounded-concurrency directory-read primitive

**File:** `src/data/storage/directoryScan.ts` (new)

**Implements:**
- `export const DIRECTORY_READ_CONCURRENCY = 8` — bounded read fan-out constant (write side uses 4; reads are cheaper, so higher default)
- `export type JsonDirectoryEntryLike = { name: string; kind: "file" | "directory" }` — directory listing entry type
- `export async function listDirectoryEntries(dir: DirectoryHandleLike): Promise<JsonDirectoryEntryLike[]>` — materialized directory listing (names + kind only, no file content read)
- `export type ReadJsonDirectoryOptions = { suffix: string; onUnreadable: "throw" | "skip"; unreadableError?: (fileName: string) => string; concurrency?: number }` — options for bounded JSON reads
- `export type ReadJsonDirectoryResult<T> = { values: T[]; fileNames: string[]; matchedNames: string[] }` — index-aligned result with matched/readable/total names
- `export async function readJsonDirectory<T>(dir: DirectoryHandleLike, options: ReadJsonDirectoryOptions): Promise<ReadJsonDirectoryResult<T>>` — read every JSON file in dir matching suffix with bounded concurrency, preserving listing order in result regardless of worker finish order. Deterministic failure ordering (lowest index) when onUnreadable==="throw" to ensure caller doesn't derive silently incomplete snapshots. Best-effort aggregation policy with "skip" mode.

**Implementation notes:**
- Uses a shared worker-pool state object (not closure `let` variables) to work around TypeScript's conservative type narrowing across async Promise.all operations.
- Materializes listing before reads start so index-assignment ordering is guaranteed.
- Sorts matched entries by name for deterministic failure reporting across runs.
- RNG seed and reservoir handling (used by Tasks 4–6) intentionally deferred — this module is the bounded-read primitive only.
