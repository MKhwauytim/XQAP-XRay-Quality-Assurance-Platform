```ts
function getDirectoryEntries(dir: DirectoryHandleLike): AsyncIterable<DirectoryEntryLike> | null {
  const d = dir as DirectoryHandleLike & {
    values?: () => AsyncIterable<DirectoryEntryLike>;
    entries?: () => AsyncIterable<[string, DirectoryEntryLike]>;
    [Symbol.asyncIterator]?: () => AsyncIterator<DirectoryEntryLike>;
  };
  if (typeof d.values === "function") return d.values.call(d);
  if (typeof d.entries === "function") {
```

```ts
}
```

```ts
type DirectoryEntryLike = { kind: "file" | "directory"; name: string };

function getDirectoryEntries(dir: DirectoryHandleLike): AsyncIterable<DirectoryEntryLike> | null {
  const candidate = dir as DirectoryHandleLike & {
    values?: () => AsyncIterable<DirectoryEntryLike>;
    entries?: () => AsyncIterable<[string, DirectoryEntryLike]>;
    [Symbol.asyncIterator]?: () => AsyncIterator<DirectoryEntryLike>;
  };
  if (candidate.values) return candidate.values();
  if (candidate.entries) {
    const entries = candidate.entries();
    return (async function* () {
      for await (const [, entry] of entries) yield entry;
    })();
  }
  if (candidate[Symbol.asyncIterator]) return candidate as AsyncIterable<DirectoryEntryLike>;
  return null;
}
```

- `export const DIRECTORY_READ_CONCURRENCY = 8` — bounded read fan-out constant (write side uses 4; reads are cheaper, so higher default)

- `export async function listDirectoryEntries(dir: DirectoryHandleLike): Promise<JsonDirectoryEntryLike[]>` — materialized directory listing (names + kind only, no file content read)
- `export type ReadJsonDirectoryOptions = { suffix: string; onUnreadable: "throw" | "skip"; unreadableError?: (fileName: string) => string; concurrency?: number }` — options for bounded JSON reads
- `export type ReadJsonDirectoryResult<T> = { values: T[]; fileNames: string[]; matchedNames: string[] }` — index-aligned result with matched/readable/total names
- `export async function readJsonDirectory<T>(dir: DirectoryHandleLike, options: ReadJsonDirectoryOptions): Promise<ReadJsonDirectoryResult<T>>` — read every JSON file in dir matching suffix with bounded concurrency. Results returned in name-sorted order (ascending, by filename) for determinism independent of concurrent worker settlement order or underlying directory API iteration order. Deterministic failure ordering (lowest-index unreadable file reported) when onUnreadable==="throw" to ensure caller doesn't derive silently incomplete snapshots. Best-effort aggregation policy with "skip" mode.

**Implementation notes:**
- Uses a shared worker-pool state object (not closure `let` variables) to work around TypeScript's conservative type narrowing across async Promise.all operations.
- Materializes listing before reads start to enable deterministic failure-index reporting.
- Sorts matched entries by filename (ascending) for deterministic failure reporting across runs, regardless of concurrent worker settlement order or directory API's unspecified iteration order.
