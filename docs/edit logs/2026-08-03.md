type FlushFn = () => void;

const pendingFlushes = new Set<FlushFn>();

export function registerPendingSaveFlush(flush: FlushFn): () => void {
  pendingFlushes.add(flush);
  return () => {
    pendingFlushes.delete(flush);
  };
}

export function __clearPendingSaveFlushesForTests(): void {
  pendingFlushes.clear();
}

function flushAll(): void {
  for (const flush of pendingFlushes) {
    try {
      flush();
    } catch {
      // Best-effort: one broken registrant must never block the others.
    }
  }
}

  document.addEventListener("visibilitychange", () => {
    if (document.visibilityState === "hidden") flushAll();

## v59.159 — 2026-08-03 — Fix (report-designer): flush pending autosave on unmount instead of silently discarding it

**What changed:** Plan7 Task 1. `EditorHost`'s autosave debounce (800ms after the last `doc` change) is scheduled by a `useEffect` scoped to `[doc]`; that effect's cleanup — which fires both when `doc` changes (about to reschedule) AND on true unmount — only ever called `clearTimeout`, discarding whatever edit was pending either way. In practice this meant an edit made less than 800ms before navigating away (clicking "رجوع", or any other unmount) was silently lost with no warning to the user. Fixed by adding a **separate** effect with an empty dependency array immediately after the existing one: its cleanup runs exactly once, on genuine unmount (unlike the `[doc]` effect's cleanup, which also fires on every keystroke), so flushing from there — rather than from the existing effect's cleanup — cannot turn into a save-on-every-keystroke regression. The flush is a no-op when the debounce has already fired naturally (`saveTimerRef.current` is nulled by the timer callback itself in that case), so no redundant double-save is introduced. Neither the existing `[doc]`-scoped effect nor `onBack`'s handler needed any change — `onBack` already causes an unmount today (the parent only renders `EditorHost` `if (view === "editor" && openDoc)`), so the new effect's cleanup covers the "رجوع" case automatically.

Added two tests to the existing `index.test.tsx` (no standalone test file for `EditorHost` existed — it has no separate export, so both drive it through the full `ReportDesigner` tree): one confirms a pending edit made <800ms before unmount is flushed (fails without the fix — `saveDesign` called 0 times instead of 1); the other confirms no redundant second save fires on unmount once the debounce has already fired naturally (passes both before and after the fix, as expected for a guard-clause regression check — recorded per Step 5 of the task brief rather than skipped).
